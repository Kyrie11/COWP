# 3) smoke 之后先看真正的数据性质，而不是只看最终 PASS/FAIL。
python - <<'PY'
import json, os
root=os.environ['SMOKE_ROOT']
paths={
 'screen':root+'/base_screen_verdict.json',
 'profile':root+'/fresh_profile_summary.json',
 'ablation':root+'/proposal_source_ablation.json',
 'model':root+'/model_support_audit.json',
 'natural':root+'/natural_support_diagnostic.json',
}
d={k:json.load(open(v,encoding='utf-8')) for k,v in paths.items()}
s=d['screen']; p=d['profile']; a=d['ablation']
print(json.dumps({
 'screen_pass':s.get('screen_pass'),
 'failed_checks':[k for k,v in (s.get('checks') or {}).items() if not v],
 'observed':s.get('observed'),
 'critical_modes':p.get('critical_selection_reference_modes'),
 'conflict_region_selection':p.get('conflict_region_selection'),
 'proposal_source_counts':a.get('proposal_source_candidate_counts'),
 'joint_route_ncf_increment':a.get('joint_route_ncf_increment'),
 'model_support_pass':d['model'].get('pass'),
 'natural_support_pass':d['natural'].get('pass', 'see diagnostic fields'),
},indent=2,ensure_ascii=False))
PY

python - <<'PY'
import json, os
root=os.environ['PROBE_ROOT']
r=json.load(open(root+'/v16_8_18_strict_verdict.json',encoding='utf-8'))  # 历史 orchestrator 文件名
b=json.load(open(root+'/base_screen_verdict.json',encoding='utf-8'))
p=json.load(open(root+'/fresh_probe_profile_summary.json',encoding='utf-8'))
a=json.load(open(root+'/proposal_source_ablation.json',encoding='utf-8'))
print(json.dumps({
 'pass':r.get('pass'),
 'recommend_full_rebuild':r.get('recommend_full_rebuild'),
 'failed_checks':r.get('failed_checks',[k for k,v in (r.get('checks') or {}).items() if not v]),
 'screen_failed':[k for k,v in (b.get('checks') or {}).items() if not v],
 'priority_ncf':(b.get('observed') or {}).get('new_priority_ncf_scene_rate'),
 'priority_eligible':(b.get('observed') or {}).get('new_priority_eligible_scene_rate'),
 'pbtr_floor':(b.get('observed') or {}).get('new_pbtr_floor'),
 'global_ncf':(b.get('observed') or {}).get('new_global_any_ncf_scene_rate'),
 'global_false_safe_floor':(b.get('observed') or {}).get('new_global_false_safe_floor'),
 'conflict_region_selection':p.get('conflict_region_selection'),
 'joint_route_candidates':(a.get('proposal_source_candidate_counts') or {}).get('JOINT_ROUTE_NCF',0),
 'joint_route_increment':a.get('joint_route_ncf_increment'),
