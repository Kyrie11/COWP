# 0. Enter the v48.4 repository.
cd /path/to/OC-RAP-v48.4-SRGR

# 1. Main v48.4 screening.  Warm-start from v48.1 precision, not v47/scratch.
OUTPUTDIR=runs/ocrap_v48_4_srgr_proxy_4801 \
TRAIN_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
EVAL_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
INIT_CKPT=runs/ocrap_v48_1_existing_data_screening/candidates/precision/model_v48_trac_sr/best.pt \
CALIBRATION_MODE=proxy_val_split CALIBRATION_FRACTION=0.50 CALIBRATION_SEED=4801 \
BUILD_TRAIN=0 BUILD_CALIBRATION=0 STRICT_TRAIN_DATA_GATE=0 REUSE_TEACHER_INDEX=0 \
GPU0=0 GPU1=1 BATCH_SIZE=72 NUM_WORKERS=6 PREFETCH_FACTOR=2 \
EPOCHS=10 PATIENCE=3 \
SET_CONTEXT_ENABLED=true POLICY_DISTILL_WEIGHT=1.0 POLICY_REGRET_WEIGHT=1.0 \
POLICY_DECOUPLE_ADMISSION=true POLICY_ADMISSION_DISTILL_WEIGHT=0.15 \
GROUP_DRO_WEIGHT=0.35 GROUP_DRO_TEMPERATURE=0.35 \
OPPORTUNITY_SOFT_LABEL_TEMPERATURE=0.02 HARM_SOFT_LABEL_TEMPERATURE=0.02 \
bash run_v48_two_gpu_fast_commands.txt

# Controller is backgrounded by default.
tail -f runs/ocrap_v48_4_srgr_proxy_4801/logs/controller.log
cat runs/ocrap_v48_4_srgr_proxy_4801/screening_status.json

# 2. Four explicit ablations. They run sequentially; each experiment trains balanced
# and precision variants concurrently on GPU0/GPU1 and uses its own output directory.
nohup env \
ABLATION_ROOT=runs/ocrap_v48_4_ablations \
TRAIN_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
EVAL_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
INIT_CKPT=runs/ocrap_v48_1_existing_data_screening/candidates/precision/model_v48_trac_sr/best.pt \
CALIBRATION_SEED=4801 GPU0=0 GPU1=1 BATCH_SIZE=72 EPOCHS=8 PATIENCE=2 \
bash scripts/run_v48_4_core_ablations.sh \
> runs/ocrap_v48_4_ablations.controller.log 2>&1 </dev/null &

tail -f runs/ocrap_v48_4_ablations.controller.log
cat runs/ocrap_v48_4_ablations/ablation_summary.json

# 3. Calibration-seed robustness with ONE fixed trained checkpoint.
# CALIBRATION_SEED is effectively 4801/4802/4803 inside the script, but every seed
# gets a separate directory. Do not point these runs at the same OUTPUTDIR.
nohup env \
SOURCE_RUN=runs/ocrap_v48_4_srgr_proxy_4801 \
MULTISEED_ROOT=runs/ocrap_v48_4_srgr_proxy_multiseed \
EVAL_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
SEEDS=4801,4802,4803 CALIBRATION_FRACTION=0.50 GPU0=0 GPU1=1 \
bash scripts/recalibrate_v48_4_multiseed.sh \
> runs/ocrap_v48_4_srgr_proxy_multiseed.controller.log 2>&1 </dev/null &

tail -f runs/ocrap_v48_4_srgr_proxy_multiseed.controller.log
cat runs/ocrap_v48_4_srgr_proxy_multiseed/multiseed_summary.json

# 4. Safe-only non-inferiority probe can be run even before Near/Contact pass.
# It does not authorize Near/Contact deployment.
BASE_RUN=runs/ocrap_v48_4_srgr_proxy_4801/candidates/precision \
RUN=runs/ocrap_v48_4_srgr_proxy_4801/safe_noninferiority_probe \
SAFE_TEST=runs/ocrap_v48_4_srgr_proxy_4801/dataset_splits/development_safe \
NEAR_TEST=runs/ocrap_v48_4_srgr_proxy_4801/dataset_splits/development_near_contact \
CONTACT_TEST=runs/ocrap_v48_4_srgr_proxy_4801/dataset_splits/development_contact \
RUN_OFFLINE_EVAL=0 RUN_AUDITS=0 RUN_SAFE_CLOSED_LOOP=1 RUN_SCALAR_BASELINES=1 \
SAFE_MAX_ROLLOUTS=32 SAFE_MAX_STEPS=24 GPU_SAFE=0 \
nohup bash scripts/run_ocrap_v48_trac_sr.sh \
> runs/ocrap_v48_4_srgr_proxy_4801/logs/safe_noninferiority_probe.log 2>&1 </dev/null &

# 5. Near/Contact closed-loop probe ONLY after both Natural gates pass.
# The controller writes this file only for a valid candidate.
bash runs/ocrap_v48_4_srgr_proxy_4801/NEXT_COMMANDS.txt
