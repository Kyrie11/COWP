# COWP v16.8.19 数据支持重建顺序
# 重要：本版本修改了 label 语义，禁止对 v16.8.16/v16.8.18 旧标签只做 reaudit。

cd /path/to/COWP_v16_8_19
export PYTHONPATH="$PWD:${PYTHONPATH:-}"
export WOMD_ROOT=/data0/senzeyu2/dataset/WOMD/waymo_open_dataset_motion_v_1_3_1
export SOURCE_DATA_ROOT=/data0/senzeyu2/dataset/COWP/formal

# 1) WOMD split / v1.3.1 / sdc_paths 数据合同
bash NEXT_EXECUTION_V16_8_19_CN.sh split-audit
bash NEXT_EXECUTION_V16_8_19_CN.sh preflight

# 2) 必须 fresh smoke（wrapper 会把 smoke 映射成 fresh-smoke）
bash NEXT_EXECUTION_V16_8_19_CN.sh smoke

# 3) 推荐：确认 risk-known-zero fast path 没改变标签语义
bash NEXT_EXECUTION_V16_8_19_CN.sh fastpath-ab

# 4) 1200-scene validation strict
bash NEXT_EXECUTION_V16_8_19_CN.sh strict

# 5) 只有 strict verdict 的 recommend_full_rebuild=true 才运行 train-pilot
python - <<'PY'
import json, os
p=os.environ.get('PROBE_ROOT','/data0/senzeyu2/dataset/COWP/formal_v16_8_19_support_strict_probe') + '/v16_8_18_strict_verdict.json'
r=json.load(open(p,encoding='utf-8'))
print(json.dumps({'pass':r.get('pass'),'recommend_full_rebuild':r.get('recommend_full_rebuild'),'checks':r.get('checks')},indent=2,ensure_ascii=False))
assert r.get('recommend_full_rebuild') is True, 'STRICT 仍未授权 train-pilot；不要跳 gate'
PY
bash NEXT_EXECUTION_V16_8_19_CN.sh train-pilot

# 6) 只有 train-pilot 也授权后才 full-core
python - <<'PY'
import json, os
p=os.environ.get('TRAIN_PILOT_ROOT','/data0/senzeyu2/dataset/COWP/formal_v16_8_19_train_pilot') + '/v16_8_18_train_pilot_verdict.json'
r=json.load(open(p,encoding='utf-8'))
print(json.dumps({'pass':r.get('pass'),'recommend_full_rebuild':r.get('recommend_full_rebuild'),'checks':r.get('checks')},indent=2,ensure_ascii=False))
assert r.get('recommend_full_rebuild') is True, 'TRAIN PILOT 仍未授权 full-core'
PY
bash NEXT_EXECUTION_V16_8_19_CN.sh full-core

# 7) core rebuild 完成后再挂 Waymax outcomes
bash NEXT_EXECUTION_V16_8_19_CN.sh outcomes

# 任意阶段查看状态
bash NEXT_EXECUTION_V16_8_19_CN.sh check
