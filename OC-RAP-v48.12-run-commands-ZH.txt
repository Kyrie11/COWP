# v48.12 TRIDENT下一步执行指令
# calibration_contact未完成前，继续统一使用proxy_val_split。
# 主实验、多seed、消融和closed-loop必须串行；消融脚本内部最多并发两个单GPU任务。

# 0. 进入目录并检查GPU/旧任务
cd /path/to/OC-RAP-v48.12-TRIDENT
export PYTHONPATH="$PWD/src${PYTHONPATH:+:$PYTHONPATH}"
ps -ef | grep -E 'run_v48|train_ocrap|calibrate_policy|closed_loop' | grep -v grep || true
nvidia-smi

# 1. 选择与variant匹配的v48.11初始化checkpoint
BAL_CKPT=runs/ocrap_v48_11_caster_proxy_4801/candidates/balanced/model_v48_trac_sr/best.pt
PREC_CKPT=runs/ocrap_v48_11_caster_proxy_4801/candidates/precision/model_v48_trac_sr/best.pt
[[ -f "$BAL_CKPT" ]] || { echo "missing $BAL_CKPT"; exit 2; }
[[ -f "$PREC_CKPT" ]] || { echo "missing $PREC_CKPT"; exit 2; }

# 2. 主实验：两张A30分别运行balanced/precision
OUTPUTDIR=runs/ocrap_v48_12_trident_proxy_4801 \
RUN_LABEL=v48.12-TRIDENT \
TRAIN_SCRIPT=scripts/train_ocrap_v48_12_trident.sh \
TRAIN_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
EVAL_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
INIT_CKPT="$BAL_CKPT" \
INIT_CKPT_BALANCED="$BAL_CKPT" \
INIT_CKPT_PRECISION="$PREC_CKPT" \
CALIBRATION_MODE=proxy_val_split \
CALIBRATION_FRACTION=0.50 \
CALIBRATION_SEED=4801 \
BUILD_TRAIN=0 BUILD_CALIBRATION=0 STRICT_TRAIN_DATA_GATE=0 \
AUTO_ENSURE_MANIFESTS=0 \
PREBUILT_SPLIT_ROOT=runs/ocrap_v48_8_shared_assets_4801/dataset_splits \
REUSE_PREBUILT_SPLITS=1 \

REUSE_TEACHER_INDEX=1 \
GPU0=0 GPU1=1 \
BATCH_SIZE=72 NUM_WORKERS=6 PREFETCH_FACTOR=2 \
PREFERENCE_EPOCHS=14 PREFERENCE_PATIENCE=4 \
EVIDENCE_EPOCHS=12 EVIDENCE_PATIENCE=4 \
PREFERENCE_CONDITIONAL_PAIRWISE_WEIGHT=1.00 \
PREFERENCE_CONDITIONAL_PAIRWISE_MIN_GAP=0.012 \
PREFERENCE_CONDITIONAL_PAIRWISE_MARGIN=0.018 \
ORDERED_EVIDENCE_PAIRWISE_BENEFIT_WEIGHT=0.60 \
ORDERED_EVIDENCE_PAIRWISE_HARM_WEIGHT=1.40 \
ORDERED_EVIDENCE_PAIRWISE_MARGIN=0.25 \
MACRO_CONSTRAINT_MODE=opportunity_normalized \
MAX_MACRO_EXCESS_SHARE=0.15 \
RISK_SOURCE=ordinal_evidence \
bash run_v48_two_gpu_fast_commands.txt

# 3. 完成性和分层学习检查
python tools/audit_v48_7_completion.py \
  --root runs/ocrap_v48_12_trident_proxy_4801 \
  --require-calibration \
  --output runs/ocrap_v48_12_trident_proxy_4801/completion_audit_v48_12.json

python tools/check_v48_12_learning_gates.py \
  --run runs/ocrap_v48_12_trident_proxy_4801 \
  --output runs/ocrap_v48_12_trident_proxy_4801/learning_gates_v48_12.json || true

cat runs/ocrap_v48_12_trident_proxy_4801/learning_gates_v48_12.json
cat runs/ocrap_v48_12_trident_proxy_4801/GATE_FAILED.json 2>/dev/null || true
cat runs/ocrap_v48_12_trident_proxy_4801/NEXT_COMMANDS.txt 2>/dev/null || true

# 4. 三seed固定checkpoint复校准
# 仅在completion audit通过后运行。即使Natural gate暂未通过，也可用于判断稳定性；
# 不得把不同seed写到同一目录。
nohup env \
SOURCE_RUN=runs/ocrap_v48_12_trident_proxy_4801 \
MULTISEED_ROOT=runs/ocrap_v48_12_trident_proxy_multiseed \

SEEDS=4801,4802,4803 \
CALIBRATION_FRACTION=0.50 \
RISK_SOURCE=ordinal_evidence \
MACRO_CONSTRAINT_MODE=opportunity_normalized \
MAX_MACRO_EXCESS_SHARE=0.15 \
GPU0=0 GPU1=1 \
bash scripts/recalibrate_v48_12_multiseed.sh \
> runs/ocrap_v48_12_trident_proxy_multiseed.controller.log 2>&1 </dev/null &

# 5. 四组核心消融，一次提交；两张A30最多同时运行两个任务
# A: 合同修复；B: recovery pair；C: bipolar evidence；D: full TRIDENT。
# 主实验或多seed仍在占GPU时不要启动。
nohup env \

INIT_CKPT_BALANCED="$BAL_CKPT" \
INIT_CKPT_PRECISION="$PREC_CKPT" \
CALIBRATION_SEED=4801 \
MACRO_CONSTRAINT_MODE=opportunity_normalized \
MAX_MACRO_EXCESS_SHARE=0.15 \
GPU0=0 GPU1=1 \
BATCH_SIZE=72 NUM_WORKERS=6 PREFETCH_FACTOR=2 \
bash scripts/run_v48_12_parallel_ablations.sh \
> runs/ocrap_v48_12_ablations.controller.log 2>&1 </dev/null &

# 消融完成性检查
cat runs/ocrap_v48_12_ablations/ABLATIONS_STATUS.json
cat runs/ocrap_v48_12_ablations/ablation_summary_v48_12.json

# 6. Safe paired non-inferiority可独立运行，不授权Near/Contact恢复动作
# 根据服务器上脚本参数填写SCENES等；建议至少100个paired scenes。
BASE_RUN=runs/ocrap_v48_12_trident_proxy_4801/candidates/balanced \
OUTPUT_ROOT=runs/ocrap_v48_12_safe_paired_noninferiority \
GPU0=0 GPU1=1 \
bash scripts/run_v48_7_safe_noninferiority.sh

# 7. Natural gate真正通过后才运行stress closed-loop
# 只执行controller生成的命令，禁止手工选择未通过gate的checkpoint。
if [[ -f runs/ocrap_v48_12_trident_proxy_4801/NEXT_COMMANDS.txt ]]; then
  cat runs/ocrap_v48_12_trident_proxy_4801/NEXT_COMMANDS.txt
  bash runs/ocrap_v48_12_trident_proxy_4801/NEXT_COMMANDS.txt
else
  echo "Natural gate未通过：不要运行Near/Contact stress closed-loop"

bash scripts/recalibrate_v48_12_on_dedicated_set.sh
