  --output "$OUT_ROOT/eval/cache_alignment_val.json"

  --cache-dir "$RAW_TRAIN_CACHE" \
  --max-scenes 2000 \

  --cache-dir "$RAW_VAL_CACHE" \
  --max-scenes 2000 \
