# OC-RAP v48.5 ECPR 执行顺序
# 所有命令在代码包根目录运行。不要同时运行主实验、multi-seed和消融。

cd /path/to/OC-RAP-v48.5-ECPR
export PYTHONPATH="$PWD/src${PYTHONPATH:+:$PYTHONPATH}"

# 0. 确认旧任务退出
ps -ef | grep -E 'run_v48|train_ocrap_v48|ocrap.cli train|calibrate_policy_risk_v48' | grep -v grep || true
nvidia-smi

# 1. 选择初始化checkpoint。
# 优先使用服务器上真正完成且仍存在的v48.4 balanced best.pt；若不存在，回退到v48.1 precision。
V484_CKPT=runs/ocrap_v48_4_srgr_proxy_4801/candidates/balanced/model_v48_trac_sr/best.pt
V481_CKPT=runs/ocrap_v48_1_existing_data_screening/candidates/precision/model_v48_trac_sr/best.pt
if [[ -f "$V484_CKPT" ]]; then INIT_CKPT="$V484_CKPT"; else INIT_CKPT="$V481_CKPT"; fi
[[ -f "$INIT_CKPT" ]] || { echo "missing init checkpoint"; exit 2; }
echo "INIT_CKPT=$INIT_CKPT"

# 2. 主实验。run_v48_two_gpu_fast_commands.txt会自行后台启动controller。
OUTPUTDIR=runs/ocrap_v48_5_ecpr_proxy_4801 \
TRAIN_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
EVAL_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
INIT_CKPT="$INIT_CKPT" \
CALIBRATION_MODE=proxy_val_split CALIBRATION_FRACTION=0.50 CALIBRATION_SEED=4801 \
BUILD_TRAIN=0 BUILD_CALIBRATION=0 STRICT_TRAIN_DATA_GATE=0 REUSE_TEACHER_INDEX=0 \
GPU0=0 GPU1=1 BATCH_SIZE=72 NUM_WORKERS=6 PREFETCH_FACTOR=2 \
EPOCHS=12 PATIENCE=4 RISK_SOURCE=delta_distribution \
EXACT_TEACHER_PCD=true SET_CONTEXT_ENABLED=true \
PREFERENCE_HEAD_ENABLED=true PREFERENCE_WEIGHT=1.50 PREFERENCE_REGRET_WEIGHT=0.50 \
PREFERENCE_TEMPERATURE=0.06 PREFERENCE_MIN_GAP=0.010 \
PREFERENCE_MARGIN=0.030 PREFERENCE_CONFIDENCE_SCALE=0.040 \
DELTA_NLL_WEIGHT=0.50 GROUP_DRO_WEIGHT=0 \
POLICY_DISTILL_WEIGHT=0 POLICY_REGRET_WEIGHT=0 POLICY_ADMISSION_DISTILL_WEIGHT=0.05 \
OPPORTUNITY_AUX_WEIGHT=0.25 HARM_W=0.20 \
bash run_v48_two_gpu_fast_commands.txt

# 查看进度
tail -f runs/ocrap_v48_5_ecpr_proxy_4801/logs/controller.log

# 3. 主实验停止后检查完整性。即使Natural gate拒绝，训练和校准仍可完整。
python tools/audit_v48_5_completion.py \
  --root runs/ocrap_v48_5_ecpr_proxy_4801 \
  --require-calibration \
  --output runs/ocrap_v48_5_ecpr_proxy_4801/completion_audit.json
cat runs/ocrap_v48_5_ecpr_proxy_4801/completion_audit.json

# 4. 固定同一checkpoint做4801/4802/4803复校准。三个seed自动使用不同目录。
# 此脚本要求TRAINING_COMPLETE.json存在，并在开始/结束校验checkpoint SHA256。
nohup env \
SOURCE_RUN=runs/ocrap_v48_5_ecpr_proxy_4801 \
MULTISEED_ROOT=runs/ocrap_v48_5_ecpr_proxy_multiseed \
EVAL_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
SEEDS=4801,4802,4803 CALIBRATION_FRACTION=0.50 GPU0=0 GPU1=1 \
bash scripts/recalibrate_v48_5_multiseed.sh \
> runs/ocrap_v48_5_ecpr_proxy_multiseed.controller.log 2>&1 </dev/null &

tail -f runs/ocrap_v48_5_ecpr_proxy_multiseed.controller.log
cat runs/ocrap_v48_5_ecpr_proxy_multiseed/multiseed_summary.json

# 5. 四组核心消融。脚本按A->B->C->D串行执行，不会并发抢GPU。
# A exact pointwise; B +ZI-NASC; C +ECPR; D full ECPR+ZI-NASC+delta NLL。
nohup env \
ABLATION_ROOT=runs/ocrap_v48_5_ablations \
TRAIN_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
EVAL_OCRAP_ROOT=/data0/senzeyu2/dataset/OCRAP \
INIT_CKPT="$INIT_CKPT" CALIBRATION_SEED=4801 \
GPU0=0 GPU1=1 BATCH_SIZE=72 NUM_WORKERS=6 PREFETCH_FACTOR=2 EPOCHS=12 PATIENCE=4 \
bash scripts/run_v48_5_core_ablations.sh \
> runs/ocrap_v48_5_ablations.controller.log 2>&1 </dev/null &

tail -f runs/ocrap_v48_5_ablations.controller.log
cat runs/ocrap_v48_5_ablations/ablation_summary_v48_5.json

# 6. 进入闭环前的硬门。
# 必须同时满足：completion comparable=true；Near/Contact top-1转正；至少一个variant有非零verify selections；
# 三seed至少2个稳定；Natural gate生成chosen_base_run.txt。
if [[ -f runs/ocrap_v48_5_ecpr_proxy_4801/chosen_base_run.txt ]]; then
  CHOSEN_BASE_RUN=$(cat runs/ocrap_v48_5_ecpr_proxy_4801/chosen_base_run.txt)
  OUTPUTDIR=runs/ocrap_v48_5_ecpr_proxy_4801
  BASE_RUN="$CHOSEN_BASE_RUN" RUN="$OUTPUTDIR/closed_loop_probe" \
  SAFE_TEST="$OUTPUTDIR/dataset_splits/dev_val_safe" \
  NEAR_TEST="$OUTPUTDIR/dataset_splits/dev_val_near_contact" \
  CONTACT_TEST="$OUTPUTDIR/dataset_splits/dev_val_contact" \
  RUN_OFFLINE_EVAL=0 RUN_AUDITS=1 RUN_SAFE_CLOSED_LOOP=1 RUN_SCALAR_BASELINES=1 \
  PAIR_BUCKETS_ON_TWO_GPUS=1 GPU_NEAR=0 GPU_CONTACT=1 GPU_SAFE=0 \
  AUDIT_MAX_ROLLOUTS=16 AUDIT_MAX_STEPS=24 AUDIT_TARGETS=32 AUDIT_LABELS=512 \
  SAFE_MAX_ROLLOUTS=16 SAFE_MAX_STEPS=32 \
  nohup bash scripts/run_ocrap_v48_trac_sr.sh \
  > "$OUTPUTDIR/logs/closed_loop_probe.log" 2>&1 </dev/null &
else
  echo "Natural gate尚未通过，不运行闭环。先分析near_miss_frontier与top-1。"
fi

# 7. 专用calibration全部构建完成后，只对最佳冻结checkpoint重新校准，不重新训练。
# 将CALIBRATION_MODE改为dedicated，并设置CALIBRATION_OCRAP_ROOT；Near/Contact未完成前继续统一使用proxy。
